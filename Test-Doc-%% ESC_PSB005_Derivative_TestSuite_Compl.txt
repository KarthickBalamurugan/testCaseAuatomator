%

%

%   TC-PSB-005-05: Setting time tolerance (settling time <500ms)
%   TC-PSB-005-06: Pending re-qualification (placeholder, not run)

%   TC-PSB-005-09: Power-up initialization (cold start)
%   TC-PSB-005-10: Derivative memory reset verification (no spike at init)

%
% OUTPUT FORMAT:
%   - PSB005_Derivative_TestResults.csv  : raw numeric results
%   - PSB005_Derivative_TestResults.docx : Word report, TEXT/TABLES ONLY.
%       No HTML is generated anywhere (the old HTML fallback / Word-COM-

%   - Every test case produces its own VISIBLE MATLAB figure window
%     (input signal(s) + DotOut response) so all "scope" outputs can be

        case 'hold'
            seg = ph.value*ones(n,1);
        case 'ramp'
            seg = ph.from + (ph.to-ph.from)*(tLocal/ph.duration);
        case 'osc'
            seg = ph.center + ph.amp*sin(2*pi*ph.freq*tLocal);
    end
    tSegs{i} = tCursor + tLocal;
    PRwSegs{i} = seg;
    phaseBounds(i,:) = [tCursor, tCursor + (n-1)*ts];
    tCursor = tCursor + n*ts;
end

%% ===================== LOCAL FUNCTIONS =====================
function [t, y] = runDerivative(mdl, tVec, PRwVec, PValidVec, SysInitVec, portPRw, portPValid, portSysInit)
    % Run simulation with external input signals
    tsPRw = timeseries(PRwVec, tVec);
    tsPRw.Name = portPRw;
    tsPValid = timeseries(logical(PValidVec), tVec);
    tsPValid.Name = portPValid;
    tsSysInit = timeseries(logical(SysInitVec), tVec);
    tsSysInit.Name = portSysInit;
    ds = Simulink.SimulationData.Dataset;
    ds = ds.addElement(tsPRw, portPRw);
    ds = ds.addElement(tsPValid, portPValid);
    ds = ds.addElement(tsSysInit, portSysInit);
    in = Simulink.SimulationInput(mdl);
    in = in.setExternalInput(ds);
    in = in.setModelParameter('StopTime', num2str(tVec(end)));
    
    out = sim(in, 'ShowProgress', 'off');
    t = out.tout;
    y = getSignal(out);
end
function y = getSignal(out)
    % Extract output signal from simulation
    if isprop(out, 'simout') || isfield(out, 'simout')
        s = out.simout;
        if isa(s, 'timeseries')
            y = s.Data;
        elseif isstruct(s) && isfield(s, 'signals')
            y = s.signals.values;
        else
            y = s;
        end
    elseif ~isempty(out.who('logsout'))
        y = out.logsout.getElement(1).Values.Data;
    else
        error('Could not locate output signal in SimulationOutput - edit getSignal().');
    end
    y = y(:);  % ensure column vector
end
function [gain_2hz, gain_8hz] = computeFreqResponse(t, y, sig_2hz, sig_8hz)
    % Compute frequency response using FFT (simplified)
    dt = mean(diff(t));
    fs = 1/dt;
    N = length(y);
    
    Y = fft(y);
    S2 = fft(sig_2hz);
    S8 = fft(sig_8hz);
    
    f = (0:N-1) * fs/N;
    
    % Find bins closest to 2Hz and 8Hz
    [~, idx2] = min(abs(f - 2));
    [~, idx8] = min(abs(f - 8));
    
    gain_2hz = abs(Y(idx2)) / abs(S2(idx2));
    gain_8hz = abs(Y(idx8)) / abs(S8(idx8));
end
function out = ternary(cond, a, b)
    % Simple ternary operator
    if cond, out = a; else, out = b; end
end

function fig = plotTestCase(t, PRwVec, PValidVec, y, tcID, titleStr, plotsDir)
    % Generic 2-subplot "scope" view used by most test cases:
    %   top    -> PRw input (left axis) with PValid overlay (right axis)
    %   bottom -> DotOut (derivative) output
    % Figure is created VISIBLE so it stays on screen in MATLAB; a PNG
    % copy is also saved under plotsDir for records (see savePlot()).
    fig = figure('Visible','on','Name',sprintf('%s: %s', tcID, titleStr),'Position',[100 100 900 600]);
    subplot(2,1,1);
    yyaxis left
    plot(t, PRwVec, 'b-', 'LineWidth',1.2);
    ylabel('PRw [bar]');
    yyaxis right
    stairs(t, double(PValidVec), 'g-', 'LineWidth',1.0);
    ylabel('PValid [0/1]'); ylim([-0.2 1.2]);
    title(sprintf('%s: %s - Input Signal', tcID, titleStr)); grid on;
    subplot(2,1,2);
    plot(t, y, 'r-', 'LineWidth',1.2);
    xlabel('Time [s]'); ylabel('DotOut [bar/s]');
    title('Measured Derivative Output'); grid on;
    savePlot(fig, ['PSB005_' strrep(tcID,'-','_')], plotsDir);
end
function relPath = savePlot(fig, name, plotsDir)
    % Saves a copy of the figure to disk for records, but leaves the
    % figure window open/visible in MATLAB (no close(fig) call) so all
    % scope outputs stay on screen for direct inspection.
    if ~exist(plotsDir,'dir'), mkdir(plotsDir); end
    relPath = fullfile(plotsDir, [name '.png']);
    exportgraphics(fig, relPath, 'Resolution', 150);
end
function generateWordReport(results, meta, ids, reportFile)
    % Generates a Word (.docx) test report using MATLAB Report Generator
    % (mlreportgen.dom). Contains ONLY text and tables -- deliberately NO
    % images/plots are embedded here, and NO HTML is generated or used
    % anywhere in this process. All scope plots are shown as live MATLAB
    % figure windows instead (see plotTestCase()/savePlot() above).
    import mlreportgen.dom.*
    [~, baseName] = fileparts(reportFile);
    d = Document(baseName, 'docx');
    % ---- Title ----
    titlePara = Paragraph('Derivative Filter (PDot) Verification Test Report');
    titlePara.Style = {Bold(true), FontSize('18pt'), Color('#1A3C6E')};
    append(d, titlePara);
    subPara = Paragraph('REQ-IB_BHMS-SC-PSB-005: TC-PSB-005-01 through TC-PSB-005-11');
    subPara.Style = {FontSize('12pt'), Color('#444444')};
    append(d, subPara);
    append(d, Paragraph(' '));
    % ---- Metadata block ----
    metaLines = {
        sprintf('Requirement: REQ-IB_BHMS-SC-PSB-005')
        sprintf('Model: derivative_TestingSuite / DerivativeFilter')
        sprintf('Generated: %s', datestr(now))
        sprintf('MATLAB: %s', version)
        sprintf('Tester: ______________     SW Version: ______________')
        };
    for i = 1:numel(metaLines)
        p = Paragraph(metaLines{i});
        p.Style = {FontSize('10pt')};
        append(d, p);
    end
    append(d, Paragraph(' '));
    % ---- Executive summary ----
    nPass    = sum(results.Result == "PASS");
    nFail    = sum(results.Result == "FAIL");
    nPending = sum(results.Result == "PENDING");
    nTotal   = height(results);
    denom    = max(nTotal - nPending, 1);
    h = Paragraph('Executive Summary');
    h.Style = {Bold(true), FontSize('14pt'), Color('#1A3C6E')};
    append(d, h);
    summaryLines = {
        sprintf('Total Metrics: %d', nTotal)
        sprintf('Passed: %d (%.1f%%)', nPass, 100*nPass/denom)
        sprintf('Failed: %d', nFail)
        sprintf('Pending: %d', nPending)
        };
    for i = 1:numel(summaryLines)
        append(d, Paragraph(summaryLines{i}));
    end
    append(d, Paragraph(' '));
    append(d, PageBreak());
    % ---- Per test case sections (iterate over ALL known test case IDs,
    % so even a test case with no automated metric, e.g. TC-06, still
    % gets its own section) ----
    for i = 1:numel(ids)
        id = ids{i};
        if ~isKey(meta, id), continue; end
        m = meta(id);
        h2 = Paragraph(sprintf('%s: %s', id, m.Title));
        h2.Style = {Bold(true), FontSize('14pt'), Color('#1A3C6E')};
        append(d, h2);
        infoLines = {
            sprintf('Objective: %s', m.Objective)
            sprintf('Input: %s', m.Input)
            sprintf('Pass Criterion: %s', m.Criterion)
            };
        for k = 1:numel(infoLines)
            append(d, Paragraph(infoLines{k}));
        end
        append(d, Paragraph(' '));
        rows = results(results.TestCaseID == id, :);
        if height(rows) == 0
            noteP = Paragraph('No automated check implemented for this test case (pending).');
            noteP.Style = {Italic(true), Color('#888888')};
            append(d, noteP);
        else
            headerRow = {'Metric','Measured','Expected','Tolerance','Result','Notes'};
            tableData = cell(height(rows)+1, 6);
            tableData(1,:) = headerRow;
            for r = 1:height(rows)
                tableData{r+1,1} = char(rows.Metric(r));
                tableData{r+1,2} = sprintf('%.4f', rows.Measured(r));
                tableData{r+1,3} = sprintf('%.4f', rows.Expected(r));
                tableData{r+1,4} = sprintf('%.4f', rows.Tolerance(r));
                tableData{r+1,5} = char(rows.Result(r));
                tableData{r+1,6} = char(rows.Notes(r));
            end
            tbl = Table(tableData);
            tbl.Style = {Border('single'), ColSep('single'), RowSep('single'), Width('100%')};
            headerRowObj = tbl.Children(1);
            headerRowObj.Style = {Bold(true), BackgroundColor('#1A3C6E'), Color('#FFFFFF')};
            for r = 1:height(rows)
                resStr = char(rows.Result(r));
                rowObj = tbl.Children(r+1);
                resultCell = rowObj.Children(5);
                if strcmp(resStr,'PASS')
                    resultCell.Style = {Bold(true), Color('#0A7A0A')};
                elseif strcmp(resStr,'FAIL')
                    resultCell.Style = {Bold(true), Color('#CC0000')};
                else
                    resultCell.Style = {Italic(true), Color('#888888')};
                end
            end
            append(d, tbl);
        end
        append(d, Paragraph(' '));
        append(d, PageBreak());
    end
    footerPara = Paragraph(['Auto-generated engineering test report. Not an approved release ' ...
        'artifact. Scope plots are shown in MATLAB figure windows and are not embedded in this document.']);
    footerPara.Style = {FontSize('9pt'), Italic(true), Color('#888888')};
    append(d, footerPara);
    close(d);
    % close(d) writes the file using the name/extension given to Document().
    fprintf('DOCX report generated successfully: %s\n', reportFile);
end
